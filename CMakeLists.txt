cmake_minimum_required(VERSION 3.25)
project(EndfieldUidMask VERSION 0.1.3 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/minhook/CMakeLists.txt")
  message(FATAL_ERROR "Run tools/bootstrap.ps1 first")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Use private static upstream MinHook" FORCE)
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(UidMask SHARED src/plugin.cpp src/managed.cpp)
target_include_directories(UidMask PRIVATE src sdk/include)
target_link_libraries(UidMask PRIVATE minhook)
set_target_properties(UidMask PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/$<CONFIG>/uid-mask")
add_custom_target(package ALL DEPENDS UidMask)
foreach(file mod.ini config.ini config-entry.lua refresh.lua icon.png zml-package.json)
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${CMAKE_CURRENT_SOURCE_DIR}/mod/${file}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/uid-mask/${file}")
endforeach()
add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
  "${CMAKE_CURRENT_SOURCE_DIR}/third_party/minhook/LICENSE.txt" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/uid-mask/MINHOOK-LICENSE.txt")
enable_testing()
add_executable(CoreTests tests/core_tests.cpp)
target_include_directories(CoreTests PRIVATE src)
add_test(NAME core COMMAND CoreTests)
add_library(GameAssembly SHARED tests/managed_fixture.cpp)
target_compile_options(GameAssembly PRIVATE /Ob0)
add_executable(HookTests tests/hook_tests.cpp)
target_include_directories(HookTests PRIVATE sdk/include)
add_test(NAME native_hooks COMMAND HookTests "$<TARGET_FILE:GameAssembly>" "$<TARGET_FILE:UidMask>" "${CMAKE_CURRENT_SOURCE_DIR}/mod")
add_test(NAME native_contract_reject COMMAND HookTests "$<TARGET_FILE:GameAssembly>" "$<TARGET_FILE:UidMask>" "${CMAKE_CURRENT_SOURCE_DIR}/mod" reject)
add_test(NAME native_buffer_layout_reject COMMAND HookTests "$<TARGET_FILE:GameAssembly>" "$<TARGET_FILE:UidMask>" "${CMAKE_CURRENT_SOURCE_DIR}/mod" layout-reject)
add_test(NAME native_registration_rollback COMMAND HookTests "$<TARGET_FILE:GameAssembly>" "$<TARGET_FILE:UidMask>" "${CMAKE_CURRENT_SOURCE_DIR}/mod" rollback)
set_tests_properties(native_hooks PROPERTIES TIMEOUT 30)
set(ZML_LUPA_PATH "" CACHE PATH "Optional already installed Lupa")
set(ZML_SERVICES_TEST_EXE "" CACHE FILEPATH "External public loader services fixture")
if(ZML_LUPA_PATH AND ZML_SERVICES_TEST_EXE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME lua_api COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/lua_tests.py"
    "${ZML_LUPA_PATH}" "${ZML_SERVICES_TEST_EXE}")
  set_tests_properties(lua_api PROPERTIES TIMEOUT 30)
endif()
